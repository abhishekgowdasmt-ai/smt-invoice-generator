Why: After accept, assignment status is 'accepted'. PostgreSQL 16 FOR UPDATE uses
UPDATE RLS policies. The old assignments_driver_update USING clause required
status = 'assigned', so start/complete/idempotent-accept could not lock the row.

-- Policy USING includes assigned and accepted; WITH CHECK unchanged
SELECT policyname, cmd, qual, with_check
FROM pg_policies
WHERE schemaname = 'public'
  AND tablename = 'booking_assignments'
  AND policyname = 'assignments_driver_update';
-- expect 1 row
-- qual must include current_driver_id() and status IN ('assigned', 'accepted')
-- with_check must be status IN ('accepted', 'rejected') with current_driver_id()

-- No other smt_driver UPDATE policies on booking_assignments
SELECT COUNT(*) FROM pg_policies
WHERE schemaname = 'public'
  AND tablename = 'booking_assignments'
  AND 'smt_driver' = ANY (roles)
  AND cmd = 'UPDATE';
-- expect 1

-- Owner/staff UPDATE policies still present
SELECT COUNT(*) FROM pg_policies
WHERE schemaname = 'public'
  AND tablename = 'booking_assignments'
  AND policyname = 'assignments_staff_update';
-- expect 1
